\begin{frame}{Decorrelazione Inter-canale: modulazione di fase}
    \begin{itemize}
        \item Per decorrelare i canali si è implementata una \emph{modulazione di fase} in sotto-bande.
        \item Indicando con $\mathbf{x}_i (k)$ le sottobande del segnale sorgente, le sottobande del segnale decorrelato inviato al canale $l$ sono date da
            \begin{equation}
		\left[\mathbf{s}_i(k)\right]_l = \left[\mathbf{x}_i(k)\right]_l\, e^{\mathrm{j}\,\alpha_i \sin\left(2\pi f_l k D / f_s\right)},
		\qquad f_l = 2\,(l-1)\ \si{\hertz}.
		\label{eq:phase_mod}
	        \end{equation}
        \item I coefficienti $\alpha_i$ sono definiti come
            \vspace{0.3cm}
            \begin{table}
                \begin{tabular}{|c | c | c | c | c | c|}
                    
                    \hline
                    Sottobanda $i$ & $0-3$ & $4$ & $5$ & $6$ & $\geq 7$ \\
                    \hline
                    $\alpha_i$ $[$Gradi$]$ & $20\si{\degree}$ & $40\si{\degree}$ & $70\si{\degree}$ & $90\si{\degree}$ & $180\si{\degree}$\\
                    \hline
                \end{tabular}
                \caption{Coefficienti di modulazione di fase al variare dell'indice di sottobanda.}
            \end{table}
    \end{itemize}
\end{frame}

\begin{frame}{Decorrelazione Intra-canale: predizione lineare}
    \begin{itemize}
        \item La modulazione di fase decorrela i diversi canali, ma non agisce sulla correlazione interna a ciascun segnale.
        \item Per ogni istante $k$ e per ogni sottobanda si raccolgono $P$ vettori nella matrice  $\mathbf{S}_i(k) = [\,\underline{\mathbf{s}}_i(k-1),\, \dots,\, \underline{\mathbf{s}}_i(k-P)\,] \in \mathbb{C}^{LK_i \times P}$.
        \item I coefficienti di predizione lineare si ottengono come
            \begin{equation}
		\mathbf{a}_i(k) = \left(\mathbf{S}_i^{\mathsf{H}}\mathbf{S}_i + \delta_{\mathrm{ap}}\mathbf{I}\right)^{-1} \mathbf{S}_i^{\mathsf{H}}\, \underline{\mathbf{s}}_i(k).
		\label{eq:pred_coeff}
	        \end{equation}
        \item Il segnale sbiancato si ottiene poi da
            \begin{equation}
		\underline{\mathbf{u}}_i(k) = \underline{\mathbf{s}}_i(k) - \mathbf{S}_i(k)\,\mathbf{a}_i(k).
		\label{eq:whitened}
	\end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{Aggiornamento dei filtri adattativi}
    \begin{itemize}
        \item In ciascuna sottobanda, l'errore di predizione è calcolato a partire dal segnale originale
        \begin{equation}
		\mathbf{e}_i(k) = \mathbf{y}_i(k) - \hat{\mathbf{H}}_i(k)\,\underline{\mathbf{s}}_i(k).
		\label{eq:errore}
	    \end{equation}
        \item Al contrario, la direzione di aggiornamento dei filtri sfrutta il vettore sbiancato
            \begin{equation}
		\hat{\mathbf{H}}_i(k+1) = \hat{\mathbf{H}}_i(k) + \mu_{\mathrm{h}}\, w_i\,
		\frac{\mathbf{e}_i(k)\,\underline{\mathbf{u}}_i^{\mathsf{H}}(k)}{\lVert\underline{\mathbf{u}}_i(k)\rVert^2 + \delta_{\mathrm{h}}}.
		\label{eq:update}
	    \end{equation}
        \item Per la valutazione dell'accuratezza della stima si è usato il \ac{nm} calcolato tra le norme di Frobenius delle matrici delle RIR\begin{equation}
		\mathrm{NM} = 20\log_{10} \frac{\lVert\mathbf{H}-\hat{\mathbf{H}}\rVert_F}{\lVert\mathbf{H}\rVert_F},
		\qquad
		\lVert\mathbf{A}\rVert_F=\sqrt{\sum_{i,j} \lvert a_{ij}\rvert^2}.
		\label{eq:nm}
	\end{equation}
    \end{itemize}
\end{frame}

\begin{frame}{Effetti della lunghezza delle RIR stimate (ctd.)}
    \begin{alertblock}{Lunghezza della RIR $K$}
        \begin{itemize}
            \item Negli ambienti reali AEC e SFC richiedono valori di $K$ elevati, poiché la porzione di \ac{rir} non modellata non può essere né cancellata né compensata.
            \item Allungando $K$ cresce la frazione di coefficienti di ampiezza ridotta, la cui stima è meno accurata: il floor si alza di conseguenza.
        \end{itemize}
    \end{alertblock}

    \begin{exampleblock}{Compromesso}
        \begin{itemize}
            \item $K$ piccolo $\rightarrow$ floor migliore, coda fuori dal modello.
            \item $K$ grande $\rightarrow$ coda inclusa, floor peggiore e costo fino a $9\times$.
        \end{itemize}
    \end{exampleblock}
\end{frame}

\section{Conclusioni}
\begin{frame}{Conclusioni e sviluppi futuri}
	\begin{exampleblock}{Conclusioni}
	\begin{itemize}
		\item Si è implementato un algoritmo adattativo in sotto bande per la stima della \ac{rir}.
		\item Tramite MATLAB si è condotta una analisi parametrica di tutte le grandezze configurabili nell'algoritmo e se ne sono analizzati gli effetti.
		\item Tramite C++ si è validata l'efficacia dell'algoritmo in contesti real-time, confrontando le \ac{rir} stimate dall'algoritmo con quelle misurate.
	\end{itemize}
	\end{exampleblock}
	\begin{block}{Sviluppi futuri}
		\begin{itemize}
		\item Tuttavia, negli scenari trattati la \ac{rir} è sempre stata assunta statica, mentre l'algoritmo nasce per il tracciamento di risposte tempo-varianti.
		\item Una maggiore ottimizzazione dell'implementazione in C++ (ad esempio, considerando la rappresentazione polifase del banco DFT) permetterebbe di tracciare \ac{rir} più lunghe.
	\end{itemize}
	\end{block}
\end{frame}
